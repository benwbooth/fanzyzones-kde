cmake_minimum_required(VERSION 3.22)
project(fanzyzones_test_input LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(KWin 6.6 REQUIRED)
find_package(KDecoration3 REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Qml Quick Widgets DBus)
add_library(fanzyzonestestinput MODULE Driver.cpp)
target_link_libraries(fanzyzonestestinput PRIVATE KWin::kwin KDecoration3::KDecoration Qt6::Qml Qt6::DBus)
install(TARGETS fanzyzonestestinput LIBRARY DESTINATION share/fanzyzones-test-input)
install(FILES qmldir DESTINATION share/fanzyzones-test-input)
