cmake_minimum_required(VERSION 3.22)
project(fanzyzones_input LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(KWin 6.6 REQUIRED)
find_package(KDecoration3 REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Qml Quick Widgets)
add_library(fanzyzonesinput MODULE InputBridge.cpp)
target_link_libraries(fanzyzonesinput PRIVATE KWin::kwin KDecoration3::KDecoration Qt6::Qml Qt6::Quick)
install(TARGETS fanzyzonesinput LIBRARY DESTINATION share/fanzyzones-kde/native)
install(FILES qmldir DESTINATION share/fanzyzones-kde/native)
